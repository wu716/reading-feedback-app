时间日志已经合并过，**不要再导出、不要再插入、不要再 COUNT `time_log_nodes`**。

新加坡库可能是旧结构。`future_actions` 是香港后来才有的表；新加坡没有这张表就跳过，**不是数据丢了**。计数和导出都先看库里实际有哪些表，没有的表跳过，不要硬查。

## 第 3 步：新加坡导出其余表（不要导时间日志）

第 2 步最后出现 `SG_COUNTS_OK` 之后，再整段粘贴。用 CSV。只导出这台库里**实际存在的列**（旧库没有 `phone_verified`、`token_version`、`action_type` 等不会报错）。没有的表打印 `SKIP`，继续下一张。

导出完先把 `ls -lh /tmp/sg-backfill` 发给协助人，**先不要 scp、不要关新加坡、不要登录香港**。

```bash
set -e
set -o pipefail
cd /opt/shuran-app/deploy/aliyun
mkdir -p /tmp/sg-backfill

pg () {
  docker compose exec -T db psql -U reading_app -d reading_feedback -v ON_ERROR_STOP=1 "$@"
}

existing_cols () {
  local table="$1"
  local prefix="$2"
  shift 2
  local in_list
  in_list=$(printf "'%s'," "$@" | sed 's/,$//')
  pg -tAc "
    SELECT string_agg('${prefix}' || quote_ident(c.column_name), ', ' ORDER BY c.ordinal_position)
    FROM information_schema.columns c
    WHERE c.table_schema = 'public'
      AND c.table_name = '${table}'
      AND c.column_name IN (${in_list})
  " | tr -d '\r' | tr -d '\n'
}

dump_csv () {
  local name="$1"
  local kind="$2"
  shift 2
  local exists cols sql
  exists=$(pg -tAc "SELECT to_regclass('public.${name}') IS NOT NULL" | tr -d '[:space:]')
  if [ "$exists" != "t" ]; then
    echo "SKIP ${name} (新加坡没有这张表，不是数据丢失)"
    return 0
  fi
  case "$kind" in
    none) cols=$(existing_cols "$name" "" "$@") ;;
    user) cols=$(existing_cols "$name" "t." "$@") ;;
    msg) cols=$(existing_cols "$name" "m." "$@") ;;
    invite) cols=$(existing_cols "$name" "t." "$@") ;;
    *) echo "ERROR unknown kind ${kind}"; exit 1 ;;
  esac
  if [ -z "$cols" ]; then
    echo "ERROR ${name} 没有可导出的列"
    exit 1
  fi
  echo "DUMP ${name}"
  echo "COLS ${name}: ${cols}"
  case "$kind" in
    none)
      sql="SELECT ${cols} FROM ${name}"
      ;;
    user)
      sql="SELECT ${cols}, u.email AS owner_email FROM ${name} t JOIN users u ON u.id = t.user_id"
      ;;
    msg)
      sql="SELECT ${cols}, u.email AS owner_email FROM ai_advice_messages m JOIN ai_advice_sessions s ON s.id = m.session_id JOIN users u ON u.id = s.user_id"
      ;;
    invite)
      if echo "$cols" | grep -q 'used_by_user_id'; then
        sql="SELECT ${cols}, u.email AS used_by_email FROM invite_codes t LEFT JOIN users u ON u.id = t.used_by_user_id"
      else
        sql="SELECT ${cols}, NULL::text AS used_by_email FROM invite_codes t"
      fi
      ;;
  esac
  docker compose exec -T db psql -U reading_app -d reading_feedback -v ON_ERROR_STOP=1 -q \
    -c 'COPY ('"${sql}"') TO STDOUT WITH (FORMAT csv, HEADER true)' \
    > "/tmp/sg-backfill/${name}.csv"
  wc -l -c "/tmp/sg-backfill/${name}.csv"
}

dump_csv users none \
  id email name real_name phone phone_verified password_hash is_active \
  created_at updated_at deleted_at plan plan_expires_at token_version

dump_csv actions user \
  id user_id book_title source_excerpt action_text tags frequency status \
  action_type duration_type target_duration_days target_frequency custom_frequency_days \
  start_date end_date created_at updated_at deleted_at

dump_csv practice_logs user \
  id user_id action_id date result notes rating attempt_number success_score \
  created_at updated_at deleted_at

dump_csv daily_todos user \
  id user_id text completed todo_date remind_time reminded_at created_at updated_at deleted_at

dump_csv daily_tasks user \
  id user_id parent_id task_date text completed note sort_order familiarity \
  estimated_minutes parallel_group action_id created_at updated_at deleted_at

dump_csv daily_schedules user \
  id user_id schedule_date designed_at created_at updated_at

dump_csv future_actions user \
  id user_id text created_at updated_at deleted_at

dump_csv reading_entries user \
  id user_id book_title content reflection duration_minutes entry_date \
  created_at updated_at deleted_at

dump_csv self_talks user \
  id user_id action_id audio_path transcript created_at updated_at deleted_at

dump_csv self_talk_playback_logs user \
  id user_id self_talk_id play_date duration_seconds loops_completed loop_mode loop_target created_at

dump_csv ai_advice_sessions user \
  id session_id user_id action_id model_type web_search_enabled is_active \
  last_message_at created_at updated_at deleted_at

dump_csv ai_advice_messages msg \
  id session_id role content thinking_process web_search_results token_count \
  model_used created_at updated_at deleted_at

dump_csv self_talk_reminder_settings user \
  id user_id is_enabled daily_reminder_enabled daily_reminder_time reminder_days \
  after_action_reminder after_new_action_reminder inactive_days_threshold \
  browser_notification email_notification reading_reminder_enabled reading_reminder_time \
  created_at updated_at

dump_csv self_talk_reminder_logs user \
  id user_id reminder_type detail triggered_at dismissed_at action_taken notification_method

dump_csv ai_call_logs user \
  id user_id kind call_date created_at

dump_csv subscriptions user \
  id user_id plan start_date end_date is_active created_at updated_at

dump_csv invite_codes invite \
  id code plan note expires_at used_at used_by_user_id created_at

UPLOAD_VOL=$(docker volume ls -q | grep -E 'uploads' | head -n 1)
if [ -n "$UPLOAD_VOL" ]; then
  docker run --rm -v "${UPLOAD_VOL}:/data" -v /tmp/sg-backfill:/backup alpine sh -c 'if [ -d /data/self_talks ]; then tar czf /backup/uploads-self-talks.tgz -C /data self_talks; echo UPLOADS_PACKED; else echo NO_SELF_TALKS_DIR; fi'
else
  echo NO_UPLOADS_VOLUME
fi

ls -lh /tmp/sg-backfill
echo SG_DUMP_OK
```

成功：最后是 `SG_DUMP_OK`；`ls` 里能看到 `users.csv`、`actions.csv` 等；每个实际导出的 csv 至少有 1 行表头。`users.csv` 不是 0 字节。

header_cols () {
  local file="$1"
  head -n 1 "$file" | tr -d '\r' | awk -F',' '{
    out = ""
    for (i = 1; i <= NF; i++) {
      gsub(/^[[:space:]]+|[[:space:]]+$/, "", $i)
      gsub(/^"/, "", $i)
      gsub(/"$/, "", $i)
      if ($i == "") continue
      if (out != "") out = out ", "
      out = out "\"" $i "\""
    }
    print out
  }'
}

load_csv () {
  local table="$1"
  local file="$2"
  local cols
  if [ ! -f "/tmp/sg-backfill/${file}" ]; then
    echo "SKIP ${table} (没有 ${file}，新加坡没有这张表，不是数据丢失)"
    return 0
  fi
  if [ ! -s "/tmp/sg-backfill/${file}" ]; then
    echo "ERROR ${file} 是空文件"
    exit 1
  fi
  cols=$(header_cols "/tmp/sg-backfill/${file}")
  if [ -z "$cols" ]; then
    echo "ERROR ${file} 表头是空的"
    exit 1
  fi
  echo "LOAD ${table}"
  echo "COLS ${table}: ${cols}"
  docker compose exec -T db psql -U reading_app -d reading_feedback -v ON_ERROR_STOP=1 \
    -c "COPY ${table} (${cols}) FROM STDIN WITH (FORMAT csv, HEADER true)" \
    < "/tmp/sg-backfill/${file}"
}
